    # ========================================================
    # GET SYSTEMS / SOLUTION NODES
    # ========================================================

    def get_systems(self, target_ip=None):
        if not self.authenticated:
            self.authenticate()

        url = f"{self.base_url}{USERS_PATH.rsplit('/users', 1)[0]}/systems"

        if target_ip:
            import ipaddress
            parsed = ipaddress.ip_address(str(target_ip).strip())
            if parsed.version != 4:
                raise ValueError(f"target_ip deve ser IPv4: {target_ip}")
            url = f"{url}?ipaddress={parsed}"

        response = self._request(
            "GET", url, headers={"Accept": "application/xml"}
        )

        if response.status_code != 200:
            raise AvayaHTTPError(
                f"GET systems falhou: HTTP {response.status_code} - "
                f"{response.text[:1000]}"
            )

        root = ET.fromstring(response.content)
        if str(root.attrib.get("status", "")).strip() != "1":
            raise AvayaHTTPError("Avaya retornou falha no GET systems.")

        systems = []
        for node in root.iter():
            tag = node.tag.split("}", 1)[-1]
            if tag != "System":
                continue

            item = dict(node.attrib)
            for child in node.iter():
                if child is node:
                    continue
                ctag = child.tag.split("}", 1)[-1]
                if len(child) == 0:
                    item[ctag] = child.text or ""

            for lan in node.iter():
                if lan.tag.split("}", 1)[-1] == "LAN":
                    if str(lan.attrib.get("id", "")).upper() == "LAN1":
                        for child in lan:
                            ctag = child.tag.split("}", 1)[-1]
                            if ctag in {"IPAddress", "IP"}:
                                item["LAN1IPAddress"] = child.text or ""

            systems.append(item)

        logger.info(
            "[AVAYA] GET systems | target_ip=%s | sistemas=%s",
            target_ip or "solution", len(systems)
        )
        return systems
